#### 模板

完全特例化,需要先提供模板

```cpp
#include <iostream>
#include <string.h>
using namespace std;
template <typename T> bool compare(T a, T b) {
  cout << "template compare" << endl;
  return a > b;
}
// 特例
template <> bool compare<const char *>(const char *a, const char *b) {
  cout << "modified compare" << endl;
  return strcmp(a, b);
}
int main() {
  compare(10, 20);
  compare("aaa", "bbb");
}
```

部分特例化

```cpp
#include <iostream>
#include <string.h>
using namespace std;

template <typename T> class Test {
public:
  Test() { cout << "template init" << endl; }
};
//针对指针的特例化
template <typename Ty> class Test<Ty *> {
public:
  Test() { cout << "pointer init" << endl; }
};

int main() {
  Test<int> t1;
  Test<int *> t2;
}
```

#### 模板的实参推演

```cpp
#include <iostream>
#include <string.h>
using namespace std;

template <typename T> class Test {
public:
  Test(T a) { cout << typeid(T).name() << endl; }
};
int sum(int a, int b) { return a + b; }
int main() {
  Test t1(111);
  int a[] = {1, 2, 3};
  Test t2(a);
  Test t3(sum); // P+F+i+i+i+E = pointer to function (int, int) -> int End
  return 0;
}

```

```cpp
#include <iostream>
using namespace std;

template <typename T> class Test {
public:
  Test(T a) { cout << typeid(T).name() << endl; }
};
class Sum {
public:
  int sum(int a, int b) { return a + b; }
};
template <typename R, typename T, typename A1, typename A2> 
// 传入函数的时候还可以细分分别得到返回值、形参、甚至类
void func(R (T::*a)(A1, A2)) {
  cout << typeid(R).name() << endl;
  cout << typeid(T).name() << endl;
  cout << typeid(A1).name() << endl;
  cout << typeid(A2).name() << endl;
}
int main() {
  func(&Sum::sum);
  Test(&Sum::sum);
  return 0;
}

```
