#### 多线程

核心头文件

```cpp
#include <thread>      // 线程管理
#include <mutex>       // 互斥锁
#include <condition_variable> // 条件变量
#include <future>      // 异步任务
#include <atomic>      // 原子操作
```

```cpp
#include <iostream>
#include <thread>
using namespace std;

void hello() {
    cout << "Hello from thread!" << endl;
}

void print(int x, const string& s) {
    cout << "Thread: " << x << ", " << s << endl;
}

int main() {
    // 1. 普通函数
    thread t1(hello);

    // 2. 带参数的函数
    thread t2(print, 42, "hello");

    // 3. Lambda
    thread t3([](int x) {
        cout << "Lambda: " << x << endl;
    }, 100);

    // 4. 成员函数
    class MyClass {
    public:
        void member(int x) { cout << "Member: " << x << endl; }
    };
    MyClass obj;
    thread t4(&MyClass::member, &obj, 50);

    // 等待线程结束（必须）
    t1.join();
    t2.join();
    t3.join();
    t4.join();

    return 0;
}
```

| 函数     | 区别                    |
| ------ | --------------------- |
| join   | 等待线程结束，阻塞             |
| detach | 分离线程后台运行，分离后线程不可控，不阻塞 |

注意：主线程结束前，所有子线程必须先结束，否则会抛出异常，同时，主线程结束，进程也会结束，即强制结束可能存在的未完成的子线程

##### 多线程需要考虑线程安全

头文件

<mutex>互斥锁

非原子操作通常需要加锁(mutex)才保证线程安全

锁+双重判断，防止一个线程持有锁修改资源时，另一个线程对资源的判断出现错误。

lock_guard和unique_lock类似于智能指针，可以自动析构防止错误的情况下持有锁导致线程死锁，lock_guard是简单的实现，只有构造和析构，不支持拷贝构造和赋值，无法作为函数参数传递，unique_lock则更加完善



#### 生产者、消费者线程模型

STL中的容器默认不是线程安全的

使用条件变量同步线程

```cpp
#include <iostream>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable> // 条件变量
#include <chrono>
#include <random>
using namespace std;

// 线程安全的队列
template<typename T>
class SafeQueue {
private:
    queue<T> q;
    mutex mtx;
    condition_variable cv;
    int max_size;  // 队列最大容量，-1 表示无限制
    
public:
    SafeQueue(int max = -1) : max_size(max) {}
    
    // 生产者入队
    void push(const T& value) {
        unique_lock<mutex> lock(mtx);
        
        // 如果队列满，等待消费者取走数据
        if (max_size > 0) {
            cv.wait(lock, [this] { 
                return q.size() < max_size; 
            });
        }
        
        q.push(value);
        cout << "[生产者] 生产: " << value << " (队列大小: " << q.size() << ")" << endl;
        cv.notify_one();  // 通知消费者
    }
    
    // 消费者出队
    T pop() {
        unique_lock<mutex> lock(mtx);
        
        // 如果队列空，等待生产者生产数据
        // cv.wait(lock)就可以实现等待并释放锁，后面lambda则是在被唤醒后
        // 额外在进行检查
        cv.wait(lock, [this] { 
            return !q.empty(); 
        });
        
        T value = q.front();
        q.pop();
        cout << "[消费者] 消费: " << value << " (队列大小: " << q.size() << ")" << endl;
        cv.notify_one();  // 通知生产者（队列可能不满）通知对象随机
        // 可以使用notify_all，收到通知后从等待回到阻塞，判断条件以及获取锁
        
        return value;
    }
    
    bool empty() const {
        lock_guard<mutex> lock(mtx);
        return q.empty();
    }
    
    size_t size() const {
        lock_guard<mutex> lock(mtx);
        return q.size();
    }
};

// 生产者线程
void producer(SafeQueue<int>& q, int id) {
    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<> dis(1, 100);
    
    for (int i = 0; i < 5; i++) {
        int value = dis(gen);
        q.push(value);
        
        // 模拟生产耗时
        this_thread::sleep_for(chrono::milliseconds(dis(gen) % 200 + 50));
    }
}

// 消费者线程
void consumer(SafeQueue<int>& q, int id) {
    random_device rd;
    mt19937 gen(rd());
    uniform_int_distribution<> dis(1, 100);
    
    for (int i = 0; i < 5; i++) {
        int value = q.pop();
        
        // 模拟消费耗时（处理数据）
        this_thread::sleep_for(chrono::milliseconds(dis(gen) % 300 + 100));
    }
}

int main() {
    cout << "=== 生产者-消费者模型 ===" << endl;
    cout << "队列最大容量: 5" << endl << endl;
    
    SafeQueue<int> q(5);  // 最多存储 5 个元素
    
    // 创建 2 个生产者，2 个消费者
    thread p1(producer, ref(q), 1);
    thread p2(producer, ref(q), 2);
    thread c1(consumer, ref(q), 1);
    thread c2(consumer, ref(q), 2);
    
    p1.join();
    p2.join();
    c1.join();
    c2.join();
    
    cout << "\n所有任务完成" << endl;
    return 0;
}
```
